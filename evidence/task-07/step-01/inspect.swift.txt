import Foundation
import PDFKit
import AppKit
let document = PDFDocument(url: URL(fileURLWithPath: "submission/design-document.pdf"))!
print("pages=\(document.pageCount)")
for i in 0..<document.pageCount {
 let page = document.page(at:i)!
 let image = page.thumbnail(of:NSSize(width:900,height:1273),for:.mediaBox)
 let bitmap = NSBitmapImageRep(data:image.tiffRepresentation!)!
 try bitmap.representation(using:.png,properties:[:])!.write(to:URL(fileURLWithPath:"reports/step71/page-\(i+1).png"))
 try (page.string ?? "").write(toFile:"reports/step71/page-\(i+1).txt",atomically:true,encoding:.utf8)
}
if document.pageCount != 4 { exit(1) }
