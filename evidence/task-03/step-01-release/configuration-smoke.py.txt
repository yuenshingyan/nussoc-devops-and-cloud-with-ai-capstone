"""Run local default, configured-policy, and invalid-startup checks for one release image."""

from datetime import UTC, datetime
from json import dumps, loads
from subprocess import run
from sys import argv

from scripts.smoke_test import docker, request_json, validate_image, wait_for_health

image = argv[1]
metadata = loads(docker("image", "inspect", image))[0]
image_id = metadata["Id"]
assert metadata["Config"]["Labels"]["org.opencontainers.image.revision"] == (
    "ada935d033cd13f85e5e4adfed2d1e9ac449f6f8"
)
assert metadata["Config"]["Labels"]["org.opencontainers.image.version"] == "0.2.0"
default = validate_image(image_id)
container = docker(
    "create", "--pull=never", "--read-only", "--cap-drop=ALL",
    "--security-opt=no-new-privileges:true", "--tmpfs=/tmp:rw,noexec,nosuid,size=16m",
    "--publish=127.0.0.1::8000", "--env=FINSECURE_MAX_TRANSACTION_AMOUNT=100",
    "--env=FINSECURE_SUPPORTED_CURRENCIES=USD", image_id,
)
try:
    docker("start", container)
    inspection = loads(docker("inspect", container))[0]
    bindings = inspection["NetworkSettings"]["Ports"]["8000/tcp"]
    assert len(bindings) == 1 and bindings[0]["HostIp"] == "127.0.0.1"
    base_url = "http://127.0.0.1:" + bindings[0]["HostPort"]
    wait_for_health(container, base_url)
    status, schema = request_json(base_url + "/openapi.json")
    assert status == 200 and schema["info"]["version"] == "0.2.0"
    payload = {
        "transaction_id": "txn_config", "sender_id": "sender", "recipient_id": "recipient",
        "amount": "100.00", "currency": "USD",
    }
    checks = []
    for name, body, expected in [
        ("inclusive_custom_limit", payload, {"valid": True, "errors": []}),
        ("over_custom_limit", {**payload, "amount": "100.01"},
         {"valid": False, "errors": ["AMOUNT_LIMIT_EXCEEDED"]}),
        ("disabled_currency", {**payload, "amount": "50", "currency": "SGD"},
         {"valid": False, "errors": ["UNSUPPORTED_CURRENCY"]}),
        ("combined_rejections", {**payload, "amount": "125.50", "currency": "SGD"},
         {"valid": False, "errors": ["AMOUNT_LIMIT_EXCEEDED", "UNSUPPORTED_CURRENCY"]}),
    ]:
        status, result = request_json(base_url + "/validate", body)
        assert (status, result) == (200, expected)
        checks.append({"case": name, "status": status, "body": result})
    assert request_json(base_url + "/health") == (200, {"status": "healthy"})
    docker("stop", "--timeout=10", container)
    state = loads(docker("inspect", container))[0]["State"]
    assert not state["Running"] and not state["OOMKilled"] and state["ExitCode"] == 0
finally:
    docker("rm", "--force", container)

invalid = docker(
    "create", "--pull=never", "--read-only", "--cap-drop=ALL",
    "--security-opt=no-new-privileges:true", "--tmpfs=/tmp:rw,noexec,nosuid,size=16m",
    "--env=FINSECURE_MAX_TRANSACTION_AMOUNT=synthetic-invalid-setting", image_id,
)
try:
    docker("start", invalid)
    invalid_exit = int(docker("wait", invalid))
    assert invalid_exit != 0
    result = run(["docker", "logs", invalid], capture_output=True, text=True, check=True, timeout=10)
    logs = result.stdout + result.stderr
    assert "Invalid application configuration:" in logs
    assert "synthetic-invalid-setting" not in logs
    assert "Application startup complete" not in logs
finally:
    docker("rm", "--force", invalid)

summary = {
    "verified_at_utc": datetime.now(UTC).isoformat(), "image_reference": image,
    "image_id": image_id, "platform": metadata["Os"] + "/" + metadata["Architecture"],
    "source_revision": metadata["Config"]["Labels"]["org.opencontainers.image.revision"],
    "default_smoke": default,
    "configured_smoke": {
        "container_id": container,
        "settings": {"FINSECURE_MAX_TRANSACTION_AMOUNT": "100",
                     "FINSECURE_SUPPORTED_CURRENCIES": "USD"},
        "http_checks": checks, "shutdown_exit_code": state["ExitCode"], "container_removed": True,
    },
    "invalid_configuration": {
        "container_id": invalid, "exit_code": invalid_exit,
        "invalid_value_not_logged": True, "container_removed": True,
    },
}
print(dumps(summary, indent=2))
