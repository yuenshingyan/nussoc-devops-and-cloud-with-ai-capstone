def command(args: list[str], timeout: int=60) -> str:
 """Retain command diagnostics without exporting private configuration contents."""
 p=run(args,capture_output=True,text=True,timeout=timeout)
 with (out/'activation-commands.txt').open('a') as f:
  f.write(('Docker inspect retained in memory; only selected fields exported.\n' if args[:2]==['docker','inspect'] else p.stdout)+p.stderr)
 assert p.returncode==0,'Command failed; inspect retained diagnostic'
 return p.stdout.strip()
def ready() -> Jenkins:
 """Wait at most 180 seconds for controller authentication and existing-agent readiness."""
 deadline=monotonic()+180
 while monotonic()<deadline:
  try:
   c=Jenkins(admin_password());a=loads(c.request('/computer/finsecure-python/api/json?tree=idle,offline'))
   if a['idle'] and not a['offline']:return c
  except Exception:pass
  sleep(2)
 raise RuntimeError('Controller/agent readiness exceeded 180 seconds')
