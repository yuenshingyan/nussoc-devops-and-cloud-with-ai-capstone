def plugins(container: str) -> dict:
 """Hash reference plugin archives without extracting or executing their contents."""
 data=run(['docker','cp',container+':/usr/share/jenkins/ref/plugins','-'],capture_output=True,check=True,timeout=90).stdout
 result={}
 with TarFile.open(fileobj=BytesIO(data)) as archive:
  for member in archive.getmembers():
   if member.isfile() and member.name.endswith('.jpi'):
    with archive.extractfile(member) as f:body=f.read()
    result[member.name]={'bytes':len(body),'sha256':sha256(body).hexdigest()}
 assert result
 return result
