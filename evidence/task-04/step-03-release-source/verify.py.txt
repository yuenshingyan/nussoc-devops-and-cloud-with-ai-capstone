def literal(data):
    text=b64encode(data).decode()
    return '['+','.join(repr(text[i:i+32000]) for i in range(0,len(text),32000))+"] .join('')"

def tar(items):
    buffer=BytesIO()
    with TarFile.open(fileobj=buffer,mode='w') as archive:
        for name,data,kind in items:
            entry=TarInfo(name);entry.type=kind;entry.size=len(data)
            if kind in [LNKTYPE,SYMTYPE]: entry.linkname='../../outside';entry.size=0
            archive.addfile(entry,BytesIO(data))
    return b64encode(compress(buffer.getvalue(),mtime=0)).decode()
