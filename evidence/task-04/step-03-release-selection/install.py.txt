"""Create only the disabled job; verify that global script approvals do not change."""

from hashlib import sha256
from json import dumps, loads
from pathlib import Path

from bootstrap import BASE_URL, ROOT, Jenkins, admin_password, command
from httpx2 import BasicAuth, Client
from release_job import create_disabled_job

j = Jenkins(admin_password())
assert not loads(j.request('/queue/api/json'))['items']
assert not loads(command(['docker','inspect','finsecure-jenkins-release-agent-1']))[0]['State']['Running']
script=(ROOT/'ops/jenkins/release-job.groovy').read_text()
with Client(auth=BasicAuth('admin',admin_password()),timeout=20,trust_env=False,follow_redirects=False) as client:
    crumb=client.get(BASE_URL+'/crumbIssuer/api/json').json()
    def approval_snapshot() -> dict:
        response=client.post(BASE_URL+'/scriptText',headers={crumb['crumbRequestField']:crumb['crumb']},data={'script':"def approval=org.jenkinsci.plugins.scriptsecurity.scripts.ScriptApproval.get(); println(groovy.json.JsonOutput.toJson([approved:approval.approvedScriptHashes.toList().sort(), pending:approval.pendingScripts.collect{it.hash}.sort()]))"})
        response.raise_for_status()
        return loads(response.text)
    before=approval_snapshot()
    record=create_disabled_job(client,script)
    after=approval_snapshot()
    assert before['approved']==after['approved'], 'Unexpected script approval change; keep job disabled'
    info=client.get(BASE_URL+'/job/finsecure-release/job/delivery/api/json').json()
    assert info['disabled'] and info['lastBuild'] is None
    record.update(global_script_approvals_unchanged=True, pending_approval_count_before=len(before['pending']),pending_approval_count_after=len(after['pending']),never_built=True,release_worker_still_stopped=True,source_sha256=sha256(script.encode()).hexdigest())
    Path('reports/release-job/installation.json').write_text(dumps(record,indent=2)+'\n')
    print(dumps(record,indent=2))
