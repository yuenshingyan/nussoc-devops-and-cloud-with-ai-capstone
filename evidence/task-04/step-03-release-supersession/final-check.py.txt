"""Read-only closure check; the release worker stays stopped."""

from datetime import UTC, datetime
from hashlib import sha256
from json import dumps, loads
from os import environ
from pathlib import Path
from subprocess import run

from bootstrap import BASE_URL, Jenkins, admin_password, command
from httpx2 import BasicAuth, Client
from verify_release import check_policy
from release_job import verify_definition

j = Jenkins(admin_password())
assert not loads(j.request('/queue/api/json'))['items']
assert not loads(j.request('/computer/finsecure-python/api/json'))['offline']
last = loads(j.request('/job/finsecure-ci/lastBuild/api/json'))
assert last['number']==34 and last['result']=='SUCCESS' and not last['building']
candidate_job=Path('reports/amd64-ci/job.txt').read_text().strip()
for job in ['finsecure-release/verify-boundaries','finsecure-release/verify-scheduling','finsecure-release/verify-unapproved','release-scheduling-source-probe',candidate_job]:
    endpoint=''.join('/job/'+part for part in job.split('/'))
    assert loads(j.request(endpoint+'/api/json'))['disabled']
state=loads(command(['docker','inspect','finsecure-jenkins-release-agent-1']))[0]['State']
assert not state['Running'] and not state['OOMKilled'] and state['ExitCode']==143
with Client(auth=BasicAuth('admin',admin_password()), timeout=20, trust_env=False, follow_redirects=False) as client:
    check_policy(client)
    definition = client.get(BASE_URL+'/job/finsecure-release/job/delivery/config.xml')
    definition.raise_for_status()
    installed_source = run(['git','show','e195899:ops/jenkins/release-job.groovy'],capture_output=True,text=True,check=True,timeout=10).stdout
    verify_definition(definition.content, installed_source)
    assert sha256(installed_source.encode()).hexdigest() == '78192f53c275bf58d8e8def12ed097c133bcff7b7c3541b2136938bbc9a800cf'
    job = client.get(BASE_URL+'/job/finsecure-release/job/delivery/api/json').json()
    assert job['disabled'] and job['lastBuild'] is None
with Client(timeout=10, trust_env=False, follow_redirects=False) as public_client:
    health=public_client.get('http://35.198.197.186/health')
    assert health.status_code==200
expected=sha256(Path('ops/jenkins/release-policy.groovy').read_bytes()).hexdigest()
output=command(['docker','exec','finsecure-jenkins-controller-1','sha256sum','/usr/share/jenkins/ref/init.groovy.d/finsecure-release-policy.groovy.override','/var/jenkins_home/init.groovy.d/finsecure-release-policy.groovy'])
assert len(output.splitlines())==2 and all(line.split()[0]==expected for line in output.splitlines())
environment={**environ,'PATH':'/opt/homebrew/share/google-cloud-sdk/bin:'+environ['PATH']}
base=['kubectl','--kubeconfig',str(Path.home()/'.config/finsecure-pay/gke/kubeconfig'),'--request-timeout=20s','-n','finsecure-pay','get']

def get(resource: str) -> dict:
    result=run([*base,resource,'-o','json'],capture_output=True,text=True,timeout=30,env=environment)
    assert result.returncode==0,'Read-only cloud inspection failed'
    return loads(result.stdout)

d=get('deployment/finsecure-pay');pods=get('pods')['items']
prior=loads(Path('evidence/task-04/step-03-release-agent/final-state.json').read_text())['application']
assert d['metadata']['uid']==prior['deployment_uid'] and d['metadata']['generation']==prior['generation']
assert sha256(dumps(d['spec'],sort_keys=True).encode()).hexdigest()==prior['spec_sha256']
assert sorted(p['metadata']['uid'] for p in pods)==prior['ready_pod_uids']
assert d['status']['availableReplicas']==2 and all(any(c['type']=='Ready' and c['status']=='True' for c in p['status']['conditions']) for p in pods)
record={'observed_at_utc':datetime.now(UTC).isoformat(),'status':'verified','policy_sha256':expected,'image_and_persisted_startup_scripts_match':True,'active_guard_verified':True,'release_worker_stopped':True,'release_worker_exit_code':143,'all_five_diagnostic_jobs_disabled':True,'candidate_job':candidate_job,'release_definition_disabled_and_exact':True,'release_job_never_built':True,'source_worker_online':True,'queue_empty':True,'normal_ci_build':34,'normal_ci_result':'SUCCESS','application_matches_prior_checkpoint':True,'available_replicas':2,'public_health_status':200,'upload_attempted':False,'deployment_attempted':False}
record.update(candidate_script_installed=False, candidate_script_sha256=sha256(Path('ops/jenkins/release-job.groovy').read_bytes()).hexdigest(), installed_script_sha256=sha256(installed_source.encode()).hexdigest())
Path('reports/release-supersession/final-state.json').write_text(dumps(record,indent=2)+'\n')
print(dumps(record,indent=2))
