"""One administrator-owned remoting transfer probe, not a downstream release build."""
from base64 import b64encode
from datetime import UTC, datetime
from hashlib import sha256
from json import dumps, loads
from pathlib import Path
from subprocess import run
from sys import executable
from time import monotonic, sleep
from uuid import uuid4

from bootstrap import BASE_URL, admin_password
from httpx2 import BasicAuth, Client
from verify_release import check_policy

base = Path('reports/release-artifacts')
control = base / 'control'
context = loads((control / 'receipt.json').read_text())['context']
remote = '/home/jenkins/agent/release-artifacts-' + uuid4().hex
container = 'finsecure-jenkins-release-agent-1'
source = Path('ops/jenkins/release-transfer.groovy').read_bytes()
tool_names = ['scripts/release_artifacts.py', 'scripts/release_archive.py', 'scripts/image_scan.py',
              'scripts/security_scan.py', 'ops/jenkins/image_archive.py']
payload = {'context':context, 'remote':remote, 'lock':b64encode((control/'uv.lock').read_bytes()).decode(),
           'project':b64encode((control/'pyproject.toml').read_bytes()).decode(),
           'tools':{name:b64encode(Path(name).read_bytes()).decode() for name in tool_names}}
(base/'remote-probe.json').write_text(dumps({'remote':remote})+'\n')
result = {'probe_kind':'administrator_owned_remoting_not_downstream_pipeline', 'registry_or_gke_credentials_bound':False,
          'publishing_attempted':False, 'deployment_attempted':False,
          'trusted_tool_sha256':{name:sha256(Path(name).read_bytes()).hexdigest() for name in tool_names},
          'transfer_script_sha256':sha256(source).hexdigest()}

def groovy_bytes(data):
    text = b64encode(data).decode()
    return '[' + ','.join(repr(text[offset:offset+32000]) for offset in range(0,len(text),32000)) + "].join('')"

def console(client, code, log):
    crumb = client.get(BASE_URL+'/crumbIssuer/api/json').json()
    response = client.post(BASE_URL+'/scriptText', data={'script':code}, headers={crumb['crumbRequestField']:crumb['crumb']})
    response.raise_for_status()
    (base/log).write_text(response.text)
    return loads(response.text)

preflight = '''
def j = jenkins.model.Jenkins.get()
assert j.queue.items.length == 0
assert j.getItemByFullName('finsecure-ci').lastBuild.number == 34
assert j.getItemByFullName('finsecure-release/delivery').disabled
assert j.getItemByFullName('finsecure-release/delivery').lastBuild == null
['finsecure-release/verify-boundaries','finsecure-release/verify-scheduling','finsecure-release/verify-unapproved',
'release-scheduling-source-probe','finsecure-ci-amd64-review-20261104041956'].each { assert j.getItemByFullName(it).disabled }
println(groovy.json.JsonOutput.toJson([status:'verified', main_build:34, delivery_disabled_unbuilt:true, queue_empty:true]))
'''
transfer_code = '''
def loader = new GroovyClassLoader(jenkins.model.Jenkins.get().pluginManager.uberClassLoader)
loader.parseClass(new String(Base64.decoder.decode('__SOURCE__'), 'UTF-8'))
def transfer = loader.loadClass('FinsecureArtifactTransfer')
def payload = new groovy.json.JsonSlurperClassic().parseText(new String(Base64.decoder.decode(__PAYLOAD__), 'UTF-8'))
def j = jenkins.model.Jenkins.get()
assert j.queue.items.length == 0
assert j.getNode('finsecure-release').toComputer().idle
assert j.getItemByFullName('finsecure-release/delivery').disabled
assert j.getItemByFullName('finsecure-release/delivery').lastBuild == null
def upstream = j.getItemByFullName(payload.context.upstream_job).getBuildByNumber(payload.context.upstream_build)
def target = new hudson.FilePath(j.getNode('finsecure-release').toComputer().channel, payload.remote)
def copied = transfer.copy(upstream, target, payload.context, Base64.decoder.decode(payload.lock),
 Base64.decoder.decode(payload.project), System.nanoTime() + 120000000000L)
payload.tools.each { name, data ->
 def path = target.child('tools/' + name)
 path.parent.mkdirs(); path.parent.chmod(0700)
 transfer.writeControl(path, Base64.decoder.decode(data))
}
assert j.queue.items.length == 0
println(groovy.json.JsonOutput.toJson([status:'transferred', receipt:copied, source_job:upstream.parent.fullName,
 source_build:upstream.number, registry_or_gke_bindings:false, upstream_code_executed:false]))
'''.replace('__SOURCE__',b64encode(source).decode()).replace('__PAYLOAD__',groovy_bytes(dumps(payload).encode()))
entry = 'from sys import path,argv;from runpy import run_module;path.insert(0,argv.pop(1));run_module("scripts.release_artifacts",run_name="__main__")'
probe = '''
from sys import path,argv
path.insert(0,argv[1])
from datetime import UTC,datetime,timedelta
from pathlib import Path
from json import dumps
from scripts.release_artifacts import verify_artifacts
root,control=Path(argv[2]),Path(argv[3])
rejected=[]
file=root/'reports/trivy.json'
original=file.read_bytes()
try:
 file.write_bytes(original+b'tampered')
 try: verify_artifacts(root,control,datetime.now(UTC))
 except ValueError as error:
  assert str(error)=='TRANSFER_HASH_OR_SIZE_MISMATCH'
  rejected.append('post-transfer-byte-tampering')
 else: raise AssertionError('tampering accepted')
finally: file.write_bytes(original)
try: verify_artifacts(root,control,datetime.now(UTC)+timedelta(days=2))
except ValueError as error:
 assert str(error)=='Database is stale or from the future'
 rejected.append('synthetic-future-release-clock')
else: raise AssertionError('stale database accepted')
print(dumps({'status':'verified','rejection_cases':rejected,'restored_inputs':verify_artifacts(root,control,datetime.now(UTC))}))
'''
started = False
try:
    with Client(auth=BasicAuth('admin',admin_password()), timeout=180, trust_env=False) as client:
        check_policy(client)
        result['preflight'] = console(client,preflight,'preflight-response.txt')
        state = run(['docker','inspect','--format','{{.State.Status}}',container],capture_output=True,text=True,check=True).stdout.strip()
        assert state == 'exited'
        started = True
        run(['docker','start',container],check=True,capture_output=True,timeout=30)
        deadline = monotonic()+90
        while monotonic()<deadline:
            response = client.get(BASE_URL+'/computer/finsecure-release/api/json',params={'tree':'offline'})
            response.raise_for_status()
            if not response.json()['offline']: break
            sleep(1)
        else: raise TimeoutError('worker did not become online')
        boundary = run([executable,'ops/jenkins/verify_release.py'],capture_output=True,text=True,timeout=45)
        (base/'worker-boundaries.txt').write_text(boundary.stdout+boundary.stderr)
        boundary.check_returncode()
        result['worker_boundaries'] = loads(boundary.stdout)
        result['transfer'] = console(client,transfer_code,'transfer-response.txt')
        checked = run(['docker','exec',container,'python','-I','-c',entry,remote+'/tools','--input',remote+'/input','--control',remote+'/control'],capture_output=True,text=True,timeout=150)
        (base/'worker-verification.txt').write_text(checked.stdout+checked.stderr)
        checked.check_returncode()
        result['worker_verification'] = loads(checked.stdout)
        tested = run(['docker','exec',container,'python','-I','-c',probe,remote+'/tools',remote+'/input',remote+'/control'],capture_output=True,text=True,timeout=150)
        (base/'worker-negative-cases.txt').write_text(tested.stdout+tested.stderr)
        tested.check_returncode()
        result['worker_negative_cases'] = loads(tested.stdout)
        cleanup = '''
def j=jenkins.model.Jenkins.get()
def root=new hudson.FilePath(j.getNode('finsecure-release').toComputer().channel,'__REMOTE__')
assert root.child('control/receipt.json').exists()
root.deleteRecursive()
assert !root.exists()
assert j.queue.items.length==0
println(groovy.json.JsonOutput.toJson([owned_probe_removed:true,queue_empty:true]))
'''.replace('__REMOTE__',remote)
        result['cleanup'] = console(client,cleanup,'cleanup-response.txt')
        result['final_job_state'] = console(client,preflight,'final-job-response.txt')
        result['status']='verified'
finally:
    if started:
        stopped=run(['docker','stop','--time','20',container],capture_output=True,text=True,timeout=30)
        (base/'worker-stop.txt').write_text(stopped.stdout+stopped.stderr)
        result['stop_returncode']=stopped.returncode
    state=run(['docker','inspect','--format','{{json .State}}',container],capture_output=True,text=True,check=True).stdout
    state=loads(state)
    result['release_worker_stopped']=not state['Running']
    result['release_worker_oom_killed']=state['OOMKilled']
    result['observed_at_utc']=datetime.now(UTC).isoformat()
    (base/'live-verification.json').write_text(dumps(result,indent=2)+'\n')
print(dumps({'status':result.get('status'),'files':result['worker_verification']['verified_files'],
             'worker_stopped':result['release_worker_stopped'],'credentials_bound':False,'publication_or_deployment':False}))
