def scm = [$class: 'GitSCM', branches: [[name: '43d6771faff9d269697f64cb70a87080d224f23c']], userRemoteConfigs: [[url: 'https://github.com/yuenshingyan/nussoc-devops-and-cloud-with-ai-capstone.git']], extensions: [[$class: 'CloneOption', timeout: 2]]]
pipeline {
    agent { label 'python' }
    options {
        skipDefaultCheckout(true)
        disableConcurrentBuilds()
        skipStagesAfterUnstable()
        timeout(time: 30, unit: 'MINUTES')
    }
    stages {
        stage('Checkout') {
            steps {
                deleteDir()
                checkout scm
                sh 'test "$(git rev-parse HEAD)" = 43d6771faff9d269697f64cb70a87080d224f23c'
                sh '''#!/usr/bin/env bash
set -euo pipefail
mkdir -p reports
{
    printf 'SOURCE_REVISION='
    git rev-parse HEAD
    python --version
    uv --version
    git --version
    sha256sum uv.lock
} > reports/build-metadata.txt
cat reports/build-metadata.txt
'''
            }
        }
        stage('Locked dependencies') {
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
uv sync --locked 2>&1 | tee reports/dependencies.txt
'''
            }
        }
        stage('Code checks') {
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
uv run --locked ruff check . 2>&1 | tee reports/lint.txt
uv run --locked ruff format --check . 2>&1 | tee reports/format.txt
'''
            }
        }
        stage('Tests') {
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
./scripts/test.sh 2>&1 | tee reports/tests.txt
'''
            }
            post {
                always {
                    script {
                        // Attempt both publishers even if one rejects a missing/malformed report.
                        try {
                            junit testResults: 'reports/junit.xml',
                                  allowEmptyResults: false,
                                  skipPublishingChecks: true
                        } finally {
                            recordCoverage tools: [[parser: 'COBERTURA', pattern: 'reports/coverage.xml']],
                                           id: 'coverage', name: 'Application coverage',
                                           enabledForFailure: true, failOnError: true,
                                           sourceCodeRetention: 'NEVER', sourceCodeEncoding: 'UTF-8',
                                           sourceDirectories: [[path: 'src/finsecure_pay']],
                                           skipPublishingChecks: true, qualityGates: []
                        }
                    }
                }
            }
        }
        stage('Security scans') {
            options { timeout(time: 5, unit: 'MINUTES') }
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
uv run --locked python scripts/security_scan.py
'''
            }
        }
        stage('Package') {
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
uv build 2>&1 | tee reports/package.txt
set -- dist/*.whl
test "$#" -eq 1 && test -s "$1"
set -- dist/*.tar.gz
test "$#" -eq 1 && test -s "$1"
sha256sum dist/*.whl dist/*.tar.gz > reports/packages.sha256
'''
                archiveArtifacts artifacts: 'dist/*.whl,dist/*.tar.gz',
                                 allowEmptyArchive: false, fingerprint: true
            }
        }
        stage('Local image export') {
            options { timeout(time: 10, unit: 'MINUTES') }
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
git show 43d6771faff9d269697f64cb70a87080d224f23c:evidence/task-05/step-02-remediation/Dockerfile.runtime-candidate > Dockerfile
python - <<'PY_DEMO_DOCKERFILE'
from pathlib import Path; p=Path('Dockerfile'); text=p.read_text(); assert text.count('ARG PYTHON_IMAGE=python:3.14.7-slim-bookworm@sha256:82bc3c539b8813ada9d68c63b40158fa002f7f33de9bf3312a3dfdc0620dff56')==1; p.write_text(text.replace('ARG PYTHON_IMAGE=python:3.14.7-slim-bookworm@sha256:82bc3c539b8813ada9d68c63b40158fa002f7f33de9bf3312a3dfdc0620dff56','ARG PYTHON_IMAGE=python:3.14.9-slim-bookworm@sha256:945c2676dc2b18e0c1f16879244122c57c0ffede60006bf3df159c507a31bcae'))
PY_DEMO_DOCKERFILE
printf '%s  Dockerfile\n' '29cfcbb1bbe188905493b8b725634a7050e5d564c43877b97b13d6e8de19c698' | sha256sum --check -
printf 'DEMO_DOCKERFILE_SHA256=29cfcbb1bbe188905493b8b725634a7050e5d564c43877b97b13d6e8de19c698\n'
mkdir -p images
revision=$(git rev-parse HEAD)
[[ "$BUILD_TAG" =~ ^[A-Za-z0-9_][A-Za-z0-9_.-]{0,127}$ ]]
image_name="finsecure-pay-ci:${BUILD_TAG}"
# Never borrow registry credentials from a persistent agent home.
export DOCKER_CONFIG
DOCKER_CONFIG=$(mktemp -d)
trap 'rm -rf -- "$DOCKER_CONFIG"' EXIT
buildctl --version > reports/buildkit-workers.txt
buildctl --addr=unix:///run/buildkit/buildkitd.sock --timeout=5 debug workers --verbose >> reports/buildkit-workers.txt
printf 'SOURCE_REVISION=%s\\nLOCAL_TAG=%s\\nPLATFORM=linux/arm64\\n' "$revision" "$image_name" > reports/image-metadata.txt
timeout --signal=TERM --kill-after=15s 9m \
    buildctl --addr=unix:///run/buildkit/buildkitd.sock --timeout=5 build \
    --frontend=dockerfile.v0 --local context=. --local dockerfile=. \
    --opt platform=linux/arm64 --opt "label:org.opencontainers.image.revision=$revision" \
    --output "type=docker,name=$image_name,dest=images/finsecure-pay.tar" \
    --metadata-file reports/buildkit-image.json --progress=plain \
    2>&1 | tee reports/image-build.txt
test -s images/finsecure-pay.tar
gzip -n -1 images/finsecure-pay.tar
sha256sum images/finsecure-pay.tar.gz > reports/image.sha256
'''
            }
        }
        stage('Image security gate') {
            steps {
                sh '''#!/usr/bin/env bash
set -euo pipefail
uv run --locked python -m scripts.image_scan \
    --revision "$(git rev-parse HEAD)" --tag "finsecure-pay-ci:${BUILD_TAG}"
'''
                // A scanner error or policy block must never reach image archival.
                archiveArtifacts artifacts: 'images/finsecure-pay.tar.gz',
                                 allowEmptyArchive: false, fingerprint: true
            }
        }
    }
    post {
        always {
            // Exact allowlist: never archive environments, credentials, caches, or the workspace.
            archiveArtifacts artifacts: 'reports/build-metadata.txt,reports/dependencies.txt,reports/lint.txt,reports/format.txt,reports/tests.txt,reports/junit.xml,reports/coverage.xml,reports/package.txt,reports/packages.sha256,reports/buildkit-workers.txt,reports/image-metadata.txt,reports/image-build.txt,reports/buildkit-image.json,reports/image.sha256,reports/security-summary.json,reports/security-tools.txt,reports/security-export.txt,reports/runtime-requirements.txt,reports/bandit.json,reports/bandit.txt,reports/pip-audit.json,reports/pip-audit.txt,reports/image-security-summary.json,reports/image-security-tools.txt,reports/image-security-export.txt,reports/image-runtime-requirements.txt,reports/trivy-db.json,reports/trivy-db.txt,reports/trivy.json,reports/trivy.txt',
                             allowEmptyArchive: true
        }
    }
}
