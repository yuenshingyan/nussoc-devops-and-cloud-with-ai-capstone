"""Build ARM64/AMD64 Alpine review candidates only; do not adopt or publish them."""
from pathlib import Path
from json import loads,dumps
from subprocess import run
from hashlib import sha256
from datetime import UTC,datetime

root=Path('reports/step52-remediation'); info=loads((root/'candidates.json').read_text())
revision=run(['git','rev-parse','HEAD'],capture_output=True,text=True,check=True).stdout.strip()
records=loads((root/'builds.json').read_text())
for arch in ['arm64','amd64']:
 key='alpine-'+arch; tag='finsecure-pay-review:step52-py3149-'+key
 assert key not in records
 assert run(['docker','image','inspect',tag],capture_output=True,timeout=15).returncode!=0
 with (root/(key+'-build.txt')).open('w') as log:
  run(['docker','build','--platform=linux/'+arch,'--progress=plain','--file',str(root/'Dockerfile.alpine-candidate'),'--build-arg','PYTHON_IMAGE='+info['python_bases']['alpine'],'--label','org.opencontainers.image.revision='+revision,'--label','org.finsecure.review.dockerfile-sha256='+info['alpine_dockerfile_sha256'],'--tag',tag,'.'],stdout=log,stderr=-2,check=True,timeout=600)
 image=loads(run(['docker','image','inspect',tag],capture_output=True,text=True,check=True,timeout=30).stdout)[0]
 assert image['Architecture']==arch and image['Os']=='linux' and image['Config']['User']=='10001:10001'
 archive=root/(key+'.tar')
 run(['docker','image','save','--output',str(archive),image['Id']],check=True,capture_output=True,timeout=120)
 run(['gzip','-n','-1',str(archive)],check=True,capture_output=True,timeout=120)
 archive=archive.with_suffix('.tar.gz')
 records[key]={'created_at_utc':datetime.now(UTC).isoformat(),'tag':tag,'image_id':image['Id'],'platform':'linux/'+arch,'base_ref':info['python_bases']['alpine'],'application_tree_revision':revision,'candidate_dockerfile_sha256':info['alpine_dockerfile_sha256'],'archive':str(archive),'archive_sha256':sha256(archive.read_bytes()).hexdigest(),'bytes':archive.stat().st_size,'published':False}
 (root/'builds.json').write_text(dumps(records,indent=2)+'\n')
 print(key,image['Id'],'built/exported locally',flush=True)
