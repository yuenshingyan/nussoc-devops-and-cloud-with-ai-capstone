"""Offline compatibility suite against candidate-installed app/runtime packages, not rebuilt app code."""
from pathlib import Path
from json import loads,dumps
from subprocess import run
from shutil import copyfile
from io import BytesIO
from tarfile import TarFile
from datetime import UTC,datetime

root=Path('reports/step52-remediation'); context=root/'compatibility-context'
context.mkdir(exist_ok=False)
files=run(['git','ls-files'],capture_output=True,text=True,check=True).stdout.splitlines()
for name in files:
 if name in ['pyproject.toml','uv.lock'] or name.startswith(('src/','tests/','scripts/','ops/jenkins/')) and name.endswith('.py'):
  target=context/name; target.parent.mkdir(parents=True,exist_ok=True); copyfile(name,target)
(context/'Dockerfile').write_text('''ARG RUNTIME_IMAGE
FROM ghcr.io/astral-sh/uv:0.9.21@sha256:15f68a476b768083505fe1dbfcc998344d0135f0ca1b8465c4760b323904f05a AS uv
FROM ${RUNTIME_IMAGE}
USER root
COPY --from=uv /uv /usr/local/bin/uv
WORKDIR /test-project
COPY . .
ENV UV_PROJECT_ENVIRONMENT=/testenv UV_PYTHON_DOWNLOADS=never UV_LINK_MODE=copy
RUN uv sync --locked --no-install-project
ENV PYTHONPATH=/opt/venv/lib/python3.14/site-packages COVERAGE_FILE=/tmp/.coverage HOME=/tmp
USER 10001:10001
ENTRYPOINT ["/bin/sh"]
''')
records=loads((root/'builds.json').read_text()); evidence={}
for arch in ['arm64','amd64']:
 key='alpine-'+arch; image=records[key]; tag='finsecure-pay-review:step52-compatibility-'+arch
 assert run(['docker','image','inspect',tag],capture_output=True,timeout=15).returncode!=0
 assert loads(run(['docker','image','inspect',image['tag']],capture_output=True,text=True,check=True).stdout)[0]['Id']==image['image_id']
 with (root/(key+'-test-image-build.txt')).open('w') as log:
  run(['docker','build','--platform=linux/'+arch,'--progress=plain','--build-arg','RUNTIME_IMAGE='+image['tag'],'--tag',tag,str(context)],stdout=log,stderr=-2,check=True,timeout=600)
 code="""set -eu
/testenv/bin/python -c 'from importlib.util import find_spec; from json import dumps; app=find_spec("finsecure_pay.main").origin; native=find_spec("pydantic_core").origin; assert app.startswith("/opt/venv/"); assert native.startswith("/opt/venv/"); print(dumps({"application_module":app,"pydantic_core_module":native}))' > /tmp/import-provenance.json
set +e
/testenv/bin/python -m pytest -p no:cacheprovider --junitxml=/tmp/junit.xml --cov=finsecure_pay --cov-report=term-missing --cov-report=xml:/tmp/coverage.xml > /tmp/tests.txt 2>&1
result=$?
tar -czf - -C /tmp tests.txt junit.xml coverage.xml import-provenance.json
exit "$result"
"""
 result=run(['docker','run','--rm','--pull=never','--platform=linux/'+arch,'--network=none','--read-only','--cap-drop=ALL','--security-opt=no-new-privileges:true','--tmpfs=/tmp:rw,noexec,nosuid,size=128m',tag,'-c',code],capture_output=True,timeout=180)
 (root/(key+'-compatibility-stderr.txt')).write_bytes(result.stderr)
 assert result.stdout,'No compatibility reports; inspect stderr'
 with TarFile.open(fileobj=BytesIO(result.stdout),mode='r:gz') as archive:
  assert {m.name for m in archive.getmembers()}=={'tests.txt','junit.xml','coverage.xml','import-provenance.json'}
  for member in archive:
   assert member.isfile()
   with archive.extractfile(member) as stream: (root/(key+'-compatibility-'+member.name)).write_bytes(stream.read())
 assert result.returncode==0,'Compatibility suite failed; inspect retained reports; no retry'
 evidence[key]={'verified_at_utc':datetime.now(UTC).isoformat(),'candidate_image_id':image['image_id'],'test_only_image':tag,'locked_dev_dependencies_added_to_separate_testenv':True,'application_not_reinstalled':True,'network_disabled':True,'read_only_root':True,'container_removed':True,'exit_code':result.returncode,'emulated':arch=='amd64'}
 (root/'compatibility.json').write_text(dumps(evidence,indent=2)+'\n')
 print(key,'offline compatibility suite passed; candidate application used',flush=True)
